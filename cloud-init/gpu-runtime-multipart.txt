#!/bin/sh
set -eu
exec >>/var/log/gpu-firstboot.log 2>&1
date -u +%FT%TZ

# If the base image lacks /etc/machine-id, systemd-networkd cannot derive
# its DHCPv6 DUID and marks even a static IPv4 link failed. Create the
# per-instance ID; never ship a populated ID in the image.
if [ ! -s /etc/machine-id ]; then
  systemd-machine-id-setup
fi
test -s /etc/machine-id

# Nova ConfigDrive is authoritative for Ethernet. Do not create another
# netplan Ethernet definition here: two names/routes for the same physical MAC
# make netplan reject the whole configuration. Reapply the already-generated
# ConfigDrive profile now that machine-id exists.
netplan generate
if ! netplan apply; then
  systemctl restart systemd-networkd
fi
ip -br address
ip route
install -d -m 0755 /var/lib/ai-build-tools
date -u +%FT%TZ >/var/lib/ai-build-tools/firstboot-complete
